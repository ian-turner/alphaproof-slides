\section{Background}

\begin{frame}{Theorem Proving}
  A \textbf{proof} is a finite sequence of logical steps that establishes a statement must be true.
  \vspace{0.8em}
  \begin{block}{Theorem}
    If $n$ is even, then $n^2$ is even.
  \end{block}
  \begin{block}{Proof}
    Let $n$ be even. Then $n = 2k$ for some integer $k$. So
    \[
      n^2 = (2k)^2 = 4k^2 = 2(2k^2),
    \]
    which is divisible by 2, hence even. \qed
  \end{block}
\end{frame}

\begin{frame}{Machine Formalization}
  Lean is built on the \textbf{Calculus of Constructions}: a dependent type theory where
  \textbf{propositions are types} and \textbf{proofs are terms} (Curry--Howard).
  \vspace{0.5em}
  \begin{columns}
    \column{0.48\textwidth}
    \begin{block}{Informal statement}
      ``$n^2$ is even whenever $n$ is even''
    \end{block}
    \begin{block}{Lean type}
      \small\ttfamily
      $\forall\, n : \mathbb{Z},\; 2 \mid n \to 2 \mid n^2$
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Dependent types}
      Types may depend on \emph{values}: $\Pi\, x{:}\alpha,\; \beta\, x$.
      This lets Lean encode rich invariants (e.g.\ \texttt{vec\,$\alpha$\,$n$}: a list of \emph{exactly} $n$ elements).
    \end{block}
  \end{columns}
  \vspace{0.5em}
  To prove a proposition $P$, construct a term of type $P$.
\end{frame}

\begin{frame}[fragile]{Proof Tree: $2 \mid n \;\Rightarrow\; 2 \mid n^2$}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{prooftree}
      \AxiomC{$[2 \mid n]^{1}$}
      \UnaryInfC{$\exists\, k \in \mathbb{Z},\; n = 2k$}
      \UnaryInfC{$n^2 = (2k)^2 = 2 \cdot (2k^2)$}
      \UnaryInfC{$\exists\, m \in \mathbb{Z},\; n^2 = 2m$}
      \UnaryInfC{$2 \mid n^2$}
      \RightLabel{${\to}$I$^{1}$}
      \UnaryInfC{$2 \mid n \;\to\; 2 \mid n^2$}
    \end{prooftree}

    \column{0.45\textwidth}
    \small
    \begin{itemize}
      \item[{$[\ ]^1$}] Temporary \textbf{assumption}, discharged at the end.
      \item[{$\exists$E}] Unpack the definition of divisibility: $2\mid n$ means $n=2k$ for some $k$.
      \item[{calc}] Pure arithmetic --- no logic rules, just algebra.
      \item[{$\exists$I}] Re-pack: exhibiting $m = 2k^2$ witnesses $2 \mid n^2$.
      \item[{${\to}$I$^1$}] \textbf{Discharge} assumption 1, having derived $2\mid n^2$
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Lean 4: Calc Proof}
\begin{minted}{lean}
theorem even_sq (n : Int) (h : 2 ∣ n) : 2 ∣ n ^ 2 :=
  let ⟨k, hk⟩ := h
  ⟨2 * k ^ 2, calc
    n ^ 2 = (2 * k) ^ 2       := by rw [hk]
        _ = 2 * k * (2 * k)   := by rw [sq]
        _ = 2 * (k * (2 * k)) := by rw [mul_assoc]
        _ = 2 * (2 * (k * k)) := by rw [mul_left_comm k 2 k]
        _ = 2 * (2 * k ^ 2)   := by rw [← sq k]⟩
\end{minted}
\vspace{0.5em}
\begin{itemize}
  \small
  \item Each step applies a single algebraic law: \texttt{sq}, \texttt{mul\_assoc}, \texttt{mul\_left\_comm}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Lean 4: Tactic Proof}
\begin{minted}{lean}
theorem even_sq (n : Int) (h : 2 ∣ n) : 2 ∣ n ^ 2 := by
  obtain ⟨k, hk⟩ := h
  exact ⟨2 * k ^ 2, by rw [hk]; ring⟩
\end{minted}
\vspace{0.5em}
\begin{itemize}
  \small
  \item \texttt{obtain ⟨k, hk⟩ := h} --- unpacks $h : 2 \mid n$ into witness $k$ and proof \texttt{hk : n = 2*k} \hfill($\exists$E)
  \item \texttt{exact ⟨2 * k\^{}2, \ldots⟩} --- provides witness $m = 2k^2$; arithmetic discharged by \texttt{ring} \hfill($\exists$I + calc)
  \item \texttt{ring} --- proves ring equalities automatically by polynomial normalisation
\end{itemize}
\end{frame}

\section{Methods}

\begin{frame}{Lean Environment}
\begin{itemize}
\item The proving process is modelled as a sequential decision making problem
\item The ``state'' is the current logical state of the proof
\item The agent acts by proposing Lean tactics directly using text strings
\item Each tactic applied yields a -1 reward, and in situations where
multiple goals branch off simultaneously the minimal reward is chosen
\item Each RL episode ends either when a proof is found or timeout occurs
\end{itemize}
\end{frame}

\begin{frame}{Prover Agent}
\begin{itemize}
\item 3-billion parameter transformer encoder-decoder
\item Input: Lean tactic state
\item Output: Policy of suggested tactics, value function of expected return
\item Tree search similar to AlphaZero is used to explore possible proof tactics
\end{itemize}
\end{frame}

\begin{frame}{Training}
\begin{itemize}
\item Multi-stage training process
\item Stage 1: pre-training on billions of tokens of math texts and code
\item Stage 2: fine-tuning on hundreds of thousands of proofs already
existing in Mathlib
\item Stage 3: self-generated experience; 1 million informal math problems
      from human-written text are formalized into Lean, and similar problems
      are synthetically generated, yielding 80 million total problems
\item Continuous improvement is shown throughout all stages, demonstrating that
      the agent is able to learn far beyond what would be possible with
      a manually curated set of training data
\end{itemize}
\end{frame}

\begin{frame}{Inference}
\begin{itemize}
\item Tree search budget is scaled up during inference
\item When search alone is not enough, the system generates
      novel problem variants, through techniques such as
      simplification or generalization, and trains on those
      in real time
\item This is known as test-time reinforcement learning (TTRL)
\end{itemize}
\end{frame}

\section{Benchmarking}

\begin{frame}{Datasets}
\begin{itemize}
\item formal-imo: 258 international math olympiad problems formalized by experts
\item miniF2F: 244 math problems from a variety of math olympiads and undergraduate courses
\item PutnamBench: 672 problems from William Lowell Putnam Mathematical Competition (1962-2025)
\end{itemize}
\end{frame}

\begin{frame}{Data Separation}
\begin{itemize}
\item In order to avoid contaminating the training data with answers to benchmark goals,
      all Lean code is explicitly filtered out before it reaches the neural network
\item Only Mathlib problems are used during fine-tuning, preventing any other formalized
      solutions from leaking into the data
\item Another process is run during the RL training loop to automatically screen out
      synthetically-generated problems which are too similar to the benchmark problems
\item This does not proclude natural-language descriptions of solutions from leaking
      into the training data, meaning that the models may still get unfair ``hints''
\end{itemize}
\end{frame}
